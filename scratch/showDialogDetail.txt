        function addChecklistItem() {
            const name = prompt("Шаг 1 из 2: Введите название нового стандарта контроля:");
            if (name && name.trim()) {
                const triggerName = name.trim();
                
                const desc = prompt("Шаг 2 из 2: Введите описание (эталонный скрипт) для этого стандарта контроля:");
                const triggerDesc = desc && desc.trim() ? desc.trim() : "Кассир должен соблюдать данный регламент обслуживания.";
                
                const c = config[currentMode];
                c.checklist.push(triggerName);
                
                // Add to Reference Scripts
                if (!c.scripts) c.scripts = [];
                c.scripts.push({
                    t: triggerName,
                    s: triggerDesc
                });
                
                // Update AI Prompt
                if (c.prompt) {
                    const count = c.checklist.length;
                    const prevCount = count - 1;
                    c.prompt = c.prompt.replace(new RegExp(`по ${prevCount} ключевым`, 'g'), `по ${count} ключевым`);
                    c.prompt = c.prompt.replace(new RegExp(`\\(${prevCount} триггеров\\)`, 'g'), `(${count} триггеров)`);
                    c.prompt += `\n${count}. ${triggerName}: ${triggerDesc}`;
                    document.getElementById('ai-prompt').value = c.prompt;
                }
                
                // Add dynamically to dialogs audits!
                dialogs.forEach(d => {
                    if (d.audit) {
                        const status = Math.random() > 0.25 ? 'green' : 'red';
                        let text = 'Выполнено';
                        let comment = 'Стандарт соблюден в полной мере.';
                        if (status === 'red') {
                            text = 'Не выполнено / пропущено';
                            comment = 'Кассир проигнорировал данный стандарт контроля.';
                        }
                        d.audit.push({
                            name: triggerName,
                            status: status,
                            text: text,
                            comment: comment
                        });
                    }
                });
                
                renderSettings();
                showToast("✅ Новый стандарт, эталонный скрипт и инструкция ИИ обновлены!");
            }
        }

        function deleteChecklistItem(index) {
            if (confirm("Вы уверены, что хотите удалить этот стандарт контроля?")) {
                const c = config[currentMode];
                const triggerName = c.checklist[index];
                c.checklist.splice(index, 1);
                
                // Remove from Reference Scripts
                if (c.scripts) {
                    c.scripts = c.scripts.filter(s => s.t !== triggerName);
                }
                
                // Update AI Prompt
                if (c.prompt) {
                    const count = c.checklist.length;
                    const prevCount = count + 1;
                    c.prompt = c.prompt.replace(new RegExp(`по ${prevCount} ключевым`, 'g'), `по ${count} ключевым`);
                    c.prompt = c.prompt.replace(new RegExp(`\\(${prevCount} триггеров\\)`, 'g'), `(${count} триггеров)`);
                    
                    const lines = c.prompt.split('\n');
                    const updatedLines = [];
                    let currentNum = 1;
                    lines.forEach(line => {
                        const match = line.match(/^(\d+)\.\s*(.*)$/);
                        if (match) {
                            const namePart = match[2];
